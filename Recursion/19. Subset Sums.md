```js
Given a list **arr** of **n** integers, return sums of all subsets in it. Output sums can be printed in **any order**.

**Example 1:**

**Input:**
n = 2
arr[] = {2, 3}
**Output:**
0 2 3 5
```

## Brute
Generating power set using bit manipulation.

#include<bits/stdc++.h>

using namespace std;

void subsetSumsHelper(int ind, vector < int > & arr, int n, vector < int > & ans, int sum) {
      if (ind == n) {
        ans.push_back(sum);
        return;
      }
      //element is picked
      subsetSumsHelper(ind + 1, arr, n, ans, sum + arr[ind]);
      //element is not picked
      subsetSumsHelper(ind + 1, arr, n, ans, sum);
    }
  vector < int > subsetSums(vector < int > arr, int n) {
    vector < int > ans;
    subsetSumsHelper(0, arr, n, ans, 0);
    sort(ans.begin(), ans.end());
    return ans;
  }
};

## Optimal
Create all the subsets using recursion using the include, exclude logic (in question no. 11). Then when a subset is created put the sum into the `ans` vector.
```cpp
#include <bits/stdc++.h>
using namespace std;

void subset(vector<int> arr, vector<int>& ans, int size, int sum = 0, int i = 0) {
    if(i >= size) {
        ans.emplace_back(sum);
        return ;
    }

    subset(arr, ans, size, sum, i+1);
        
    sum += arr[i];
    subset(arr, ans, size, sum, i+1);
}

int main() {
	vector<int> ans;

    return ans;
}
```
Time Complexity: 2^n
Space Complexity: 2^n
